on run {input, parameters}
    if (count of input) is 0 then
        display dialog "No files were selected." buttons {"OK"} default button "OK"
        return
    end if

    -- Each workbook becomes a folder of CSVs (one per sheet) next to the .xlsx
    set pathArgs to ""
    repeat with selectedFile in input
        set pathArgs to pathArgs & " " & quoted form of POSIX path of selectedFile
    end repeat

    set shellScript to "echo '=== Splitting xlsx into CSVs ===' && /Users/stanleytan/anaconda3/bin/python3 /Users/stanleytan/documents/technical/python/textToSpeech/xlsx_to_csvs.py" & pathArgs & " ; "

    tell application "Terminal"
        activate
        do script shellScript in front window
    end tell
end run
