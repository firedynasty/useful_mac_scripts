on run {input, parameters}
    if (count of input) is 0 then
        display dialog "No files were selected." buttons {"OK"} default button "OK"
        return
    end if

    -- Ask: single entry or batch .txt file?
    set modeChoice to button returned of (display dialog "Create MP3 loop(s) from a single clip or a .txt file?" buttons {"Cancel", "Single", "Use .txt File"} default button "Single")

    if modeChoice is "Single" then
        set rawInput to text returned of (display dialog "Start time, End time, [Times to loop — optional, default 1]" & return & "e.g.  1:30, 1:44   or   1:30, 1:44, 3" default answer "1:30, 1:44" buttons {"Cancel", "OK"} default button "OK")
        set segmentList to {rawInput}
    else
        -- Let user pick a .txt file
        set txtFile to POSIX path of (choose file with prompt "Select a .txt file (one clip per line: start, end, [repeat]):" of type {"public.plain-text"})
        set fileContents to do shell script "cat " & quoted form of txtFile
        -- Split into lines (do shell script returns \r, not \n)
        set AppleScript's text item delimiters to return
        set allLines to text items of fileContents
        set AppleScript's text item delimiters to ""
        -- Filter blank lines
        set segmentList to {}
        repeat with aLine in allLines
            set trimmed to aLine as text
            -- Also skip comment lines starting with #
            if length of trimmed > 0 and character 1 of trimmed is not "#" then
                set end of segmentList to trimmed
            end if
        end repeat
        if (count of segmentList) is 0 then
            display dialog "The .txt file contained no valid lines." buttons {"OK"} default button "OK"
            return
        end if
    end if

    set shellScript to ""

    repeat with selectedFile in input
        set filePath to POSIX path of selectedFile
        set fileDir to do shell script "dirname " & quoted form of filePath

        repeat with segment in segmentList
            set AppleScript's text item delimiters to ","
            set parts to text items of (segment as text)
            set AppleScript's text item delimiters to ""

            if (count of parts) < 2 or (count of parts) > 3 then
                -- skip malformed lines silently
            else
                -- Trim leading/trailing whitespace from each field
                set startTime to do shell script "echo " & quoted form of (item 1 of parts as text) & " | xargs"
                set endTime to do shell script "echo " & quoted form of (item 2 of parts as text) & " | xargs"
                if (count of parts) is 3 then
                    set repeatCount to do shell script "echo " & quoted form of (item 3 of parts as text) & " | xargs"
                else
                    set repeatCount to "1"
                end if

                set shellScript to shellScript & "cd " & quoted form of fileDir & " && "
                set shellScript to shellScript & "printf " & quoted form of (startTime & "
" & endTime & "
" & repeatCount & "
") & " | "
                set shellScript to shellScript & "/Users/stanleytan/anaconda3/bin/python3 /Users/stanleytan/Documents/technical/python/audio_tools/mix_mp3.py " & quoted form of filePath & " ; "
            end if
        end repeat
    end repeat

    if shellScript is "" then
        display dialog "No valid segments found. Check that each line has format:  start, end  or  start, end, count" buttons {"OK"} default button "OK"
        return
    end if

    tell application "Terminal"
        activate
        do script shellScript in front window
    end tell
end run
