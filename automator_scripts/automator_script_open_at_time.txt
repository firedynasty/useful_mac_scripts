on run {input, parameters}
    if (count of input) is 0 then
        display dialog "No files were selected." buttons {"OK"} default button "OK"
        return
    end if

    set timeInput to text returned of (display dialog "Jump to time (e.g. 38:00 or 1:02:30 or 2280):" default answer "0:00" buttons {"Cancel", "OK"} default button "OK")

    -- Convert MM:SS or HH:MM:SS or plain seconds to integer seconds
    set seekSeconds to (do shell script "echo " & quoted form of timeInput & " | awk -F: '{if(NF==3) print int($1*3600+$2*60+$3); else if(NF==2) print int($1*60+$2); else print int($1)}'") as integer

    repeat with selectedFile in input
        set filePath to POSIX path of selectedFile

        -- Open file in VLC
        do shell script "open -a VLC " & quoted form of filePath

        -- Wait for VLC to load the file before seeking
        delay 1.5

        tell application "VLC"
            set current time to seekSeconds
            play
        end tell
    end repeat
end run
