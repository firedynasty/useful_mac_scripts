on run {input, parameters}
    if (count of input) > 0 then
        set selectedItem to POSIX path of (item 1 of input)

        set fileInfo to do shell script "if [ -d " & quoted form of selectedItem & " ]; then echo 'dir'; else echo 'file'; fi"

        if fileInfo is "dir" then
            set targetDir to selectedItem
        else
            set targetDir to do shell script "dirname " & quoted form of selectedItem
        end if

        set portResult to display dialog "Enter port number:" default answer "8000" buttons {"Cancel", "OK"} default button "OK"
        if button returned of portResult is "Cancel" then return
        set portNum to text returned of portResult

        set shellScript to "cd " & quoted form of targetDir & " && " & ¬
            "cp /Users/stanleytan/documents/technical/github/vercel_textviewer/phpAppForBluehost/index.php . && " & ¬
            "echo 'Serving: " & targetDir & "' && " & ¬
            "echo 'http://localhost:" & portNum & "' && " & ¬
            "open http://localhost:" & portNum & " && " & ¬
            "php -S localhost:" & portNum

        tell application "Terminal"
            activate
            do script shellScript
        end tell
    else
        display dialog "No folder was selected." buttons {"OK"} default button "OK"
    end if
end run
