on run {input, parameters}
    if (count of input) is 0 then
        display dialog "No files were selected." buttons {"OK"} default button "OK"
        return
    end if

    set shellScript to ""

    repeat with selectedFile in input
        set filePath to POSIX path of selectedFile

        set shellScript to shellScript & ¬
            "/Users/stanleytan/anaconda3/bin/python3 " & ¬
            "/Users/stanleytan/Documents/technical/python/create_subtitles/addbulletpoints_tosentences.py " & ¬
            "-i " & quoted form of filePath & ¬
            " && echo '✅ Done!' || echo '❌ Conversion failed' ; "
    end repeat

    tell application "Terminal"
        activate
        do script shellScript in front window
    end tell
end run
