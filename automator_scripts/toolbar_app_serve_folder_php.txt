-- Serve front Finder window folder over PHP localhost
-- App version of the Quick Action service: takes the POSIX path of the
-- selected Finder item (or the front Finder window if nothing is
-- selected) and starts `php -S` there in Terminal.
--
-- Save as an app: Script Editor > open this file > File > Export >
-- File Format: Application. Then drag the .app onto the Finder toolbar
-- (hold Cmd while dragging on newer macOS). Or via terminal:
--   osacompile -o "Serve Folder via PHP.app" posix_path_to_serve.txt

on run
	-- Determine target directory from selection or active Finder window
	tell application "Finder"
		set sel to selection
		if (count of sel) > 0 then
			set targetItem to item 1 of sel as alias
			set itemClass to class of item targetItem
			if itemClass is folder or itemClass is disk then
				set targetDir to POSIX path of targetItem
			else
				set parentFolder to (container of targetItem) as alias
				set targetDir to POSIX path of parentFolder
			end if
		else
			try
				set currentFolder to (target of front window) as alias
				set targetDir to POSIX path of currentFolder
			on error
				display dialog "No folder was selected and no Finder window is open." buttons {"OK"} default button "OK"
				return
			end try
		end if
	end tell

	-- Strip trailing slash
	if targetDir ends with "/" then
		set targetDir to text 1 thru -2 of targetDir
	end if

	set portResult to display dialog "Enter port number:" default answer "8000" buttons {"Cancel", "OK"} default button "OK"
	if button returned of portResult is "Cancel" then return
	set portNum to text returned of portResult

	set shellScript to "cd " & quoted form of targetDir & " && " & ¬
		"cp /Users/stanleytan/documents/technical/github/vercel_textviewer/phpAppForBluehost/index.php . && " & ¬
		"echo 'Serving: " & targetDir & "' && " & ¬
		"echo 'http://localhost:" & portNum & "' && " & ¬
		"open http://localhost:" & portNum & " && " & ¬
		"php -S localhost:" & portNum & "; rm -f index.php && echo 'removed index.php'"

	tell application "Terminal"
		activate
		do script shellScript
	end tell
end run
